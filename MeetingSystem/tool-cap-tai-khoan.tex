            function renderUserTable() {
                const filter = (userSearch.value || '').toLowerCase().trim();

                const roleOrder = { 'admin': 1, 'truong_to': 2, 'thu_ky': 3, 'giao_vien': 4 };
                const sorted = [...currentUserList].sort((a, b) => {
                    const oa = roleOrder[a.role] || 99;
                    const ob = roleOrder[b.role] || 99;
                    if (oa !== ob) return oa - ob;
                    return (a.displayName || '').localeCompare(b.displayName || '');
                });

                const filtered = sorted.filter(u => {
                    if (!filter) return true;
                    const name = (u.displayName || '').toLowerCase();
                    const mail = (u.email || '').toLowerCase();
                    return name.includes(filter) || mail.includes(filter);
                });

                if (filtered.length === 0) {
                    userTableBody.innerHTML = `
                        <tr class="empty-row">
                            <td colspan="5">
                                ${currentUserList.length === 0
                                    ? '📭 Chưa có tài khoản nào. Hãy tạo tài khoản đầu tiên ở trên.'
                                    : '🔍 Không tìm thấy tài khoản phù hợp với từ khóa.'}
                            </td>
                        </tr>
                    `;
                    return;
                }

                userTableBody.innerHTML = filtered.map(u => {
                    const roleLabel = getRoleLabel(u.role);
                    const roleClass = u.role || 'giao_vien';
                    const teamLabel = u.teamId ? getTeamLabel(u.teamId) : '';
                    return `
                        <tr>
                            <td>
                                <div style="font-weight:600;color:#0f172a;">${escapeHtml(u.displayName || '(Chưa có tên)')}</div>
                                <div class="uid-text">${escapeHtml((u.uid || '').substring(0, 10))}…</div>
                            </td>
                            <td>${escapeHtml(u.email || '')}</td>
                            <td>
                                <span class="role-badge ${roleClass}">${escapeHtml(roleLabel)}</span>
                            </td>
                            <td>
                                ${teamLabel ? `<span class="team-tag">${escapeHtml(teamLabel)}</span>` : '<span style="color:#94a3b8;">—</span>'}
                            </td>
                            <td class="actions-cell">
                                <button class="btn btn-warning btn-xs" onclick="openEditModal('${u.uid}')">
                                    <i class="fas fa-edit"></i> Sửa
                                </button>

                                    <i class="fas fa-trash-alt"></i> Xóa
                                </button>
                            </td>

                }).join('');
            }

            function handleExcelFile(file) {
                if (!file) return;

                const lower = file.name.toLowerCase();
                if (!lower.endsWith('.xlsx') && !lower.endsWith('.xls')) {
                    showToast('Vui lòng chọn file Excel (.xlsx hoặc .xls).', 'error');
                    return;
                }

                excelFileName.textContent = '📄 ' + file.name;

                const reader = new FileReader();
                reader.onload = function(evt) {
                    try {
                        const data = new Uint8Array(evt.target.result);
                        const workbook = XLSX.read(data, { type: 'array' });
                        const firstSheetName = workbook.SheetNames[0];
                        const firstSheet = workbook.Sheets[firstSheetName];

                        const rows = XLSX.utils.sheet_to_json(firstSheet, { defval: '' });

                        if (rows.length === 0) {
                            showToast('File Excel rỗng hoặc không có dữ liệu.', 'warning');
                            resetBulkImport();
                            return;
                        }

                        const parsed = rows.map((r, i) => {
                            const rawRole = String(r.role || '').trim();
                            const normalized = normalizeRoleValue(rawRole);
                            return {
                                _rowIndex: i + 2,
                                email: String(r.email || '').trim(),
                                password: String(r.password || '').trim(),
                                name: String(r.name || '').trim(),
                                roleRaw: rawRole,
                                role: normalized,
                                teamId: String(r.teamId || '').trim(),
                                _status: 'pending',
                                _message: ''
                            };
                        });

                        parsed.forEach(p => {
                            if (!p.email) {
                                p._status = 'invalid';
                                p._message = 'Thiếu email';
                                return;
                            }
                            if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(p.email)) {
                                p._status = 'invalid';
                                p._message = 'Email không hợp lệ';
                                return;
                            }
                            if (!p.password) {
                                p._status = 'invalid';
                                p._message = 'Thiếu password';
                                return;
                            }
                            if (p.password.length < 6) {
                                p._status = 'invalid';
                                p._message = 'Password < 6 ký tự';
                                return;
                            }
                            if (!p.role) {
                                p._status = 'invalid';
                                p._message = 'Vai trò không hợp lệ: ' + (p.roleRaw || '(trống)');
                                return;
                            }
                            p._status = 'valid';
                            p._message = 'Hợp lệ';
                        });

                        currentBulkRows = parsed;
                        renderBulkPreview();
                        showToast(`Đã đọc ${parsed.length} dòng từ file Excel.`, 'success');
                    } catch (err) {
                        console.error('Lỗi đọc file Excel:', err);
                        showToast('Không đọc được file Excel: ' + err.message, 'error');
                        resetBulkImport();
                    }
                };
                reader.onerror = function() {
                    showToast('Lỗi đọc file. Vui lòng thử lại.', 'error');
                };
                reader.readAsArrayBuffer(file);
            }

            function renderBulkPreview() {
                if (currentBulkRows.length === 0) {
                    bulkPreviewWrapper.classList.add('hidden');
                    return;
                }

                const validCount = currentBulkRows.filter(r => r._status === 'valid').length;
                const invalidCount = currentBulkRows.filter(r => r._status === 'invalid').length;

                bulkRowCount.textContent = currentBulkRows.length;
                bulkValidCount.textContent = validCount;
                bulkInvalidCount.textContent = invalidCount;
                bulkStartCount.textContent = validCount;

                bulkPreviewBody.innerHTML = currentBulkRows.map(r => {
                    let statusCls = 'pending';
                    let statusText = r._message || '—';
                    if (r._status === 'valid') { statusCls = 'valid'; statusText = '✅ Hợp lệ'; }
                    else if (r._status === 'invalid') { statusCls = 'invalid'; statusText = '❌ ' + r._message; }
                    else if (r._status === 'success') { statusCls = 'success'; statusText = '✅ Thành công'; }
                    else if (r._status === 'error') { statusCls = 'error'; statusText = '❌ ' + r._message; }
                    else if (r._status === 'processing') { statusCls = 'processing'; statusText = '⏳ Đang tạo...'; }

                    return `
                        <tr data-bulk-row="${r._rowIndex}">
                            <td style="text-align:center;font-family:monospace;">${r._rowIndex}</td>
                            <td>${escapeHtml(r.email || '—')}</td>
                            <td style="font-family:monospace;color:#64748b;">${r.password ? '•'.repeat(Math.min(r.password.length, 12)) : '—'}</td>
                            <td>${escapeHtml(r.name || '—')}</td>
                            <td>${escapeHtml(roleDisplayLabel(r.role))}</td>
                            <td>${r.teamId ? `<span class="team-tag">${escapeHtml(r.teamId)}</span>` : '<span style="color:#94a3b8;">—</span>'}</td>
                            <td><span class="status-cell ${statusCls}">${statusText}</span></td>
                        </tr>
                    `;
                }).join('');

                bulkPreviewWrapper.classList.remove('hidden');
                btnStartBulkCreate.disabled = validCount === 0;
            }

            function updatePreviewRow(rowIndex, status, message) {
                const tr = bulkPreviewBody.querySelector(`tr[data-bulk-row="${rowIndex}"]`);
                if (!tr) return;
                const cell = tr.querySelector('.status-cell');
                if (!cell) return;

                let cls = 'pending';
                let text = message || '—';
                if (status === 'success') { cls = 'success'; text = '✅ Thành công'; }
                else if (status === 'error') { cls = 'error'; text = '❌ ' + message; }
                else if (status === 'processing') { cls = 'processing'; text = '⏳ Đang tạo...'; }
                else if (status === 'invalid') { cls = 'invalid'; text = '❌ ' + message; }
                else if (status === 'valid') { cls = 'valid'; text = '✅ Hợp lệ'; }
                cell.className = 'status-cell ' + cls;
                cell.textContent = text;
            }

            async function startBulkCreate() {
                if (isBulkProcessing) return;
                if (!isAdminLoggedIn) {
                    showToast('Bạn cần đăng nhập Admin trước.', 'error');
                    return;
                }
                if (!secondaryApp) {
                    showToast('Không tìm thấy SecondaryApp. Vui lòng tải lại trang.', 'error');
                    return;
                }

                const validRows = currentBulkRows.filter(r => r._status === 'valid');
                if (validRows.length === 0) {
                    showToast('Không có dòng hợp lệ nào để tạo.', 'warning');
                    return;
                }

                const confirmMsg = `Bạn sắp tạo ${validRows.length} tài khoản.\n\n` +
                    `Thời gian dự kiến: ~${Math.ceil(validRows.length * 1)} giây.\n` +
                    `Vui lòng KHÔNG đóng tab này trong khi chạy.\n\nTiếp tục?`;
                if (!confirm(confirmMsg)) return;

                isBulkProcessing = true;
                btnStartBulkCreate.disabled = true;
                btnClearPreview.disabled = true;
                btnDownloadTemplate.disabled = true;
                excelDropZone.style.pointerEvents = 'none';
                excelDropZone.style.opacity = '0.5';

                bulkProgressWrapper.classList.remove('hidden');
                bulkLog.innerHTML = '';
                bulkProgressBar.className = 'progress-bar-inner';
                bulkProgressBar.style.width = '0%';

                logBulk('🚀 Bắt đầu tạo tài khoản hàng loạt...', 'info');
                logBulk(`📊 Tổng số: ${validRows.length} tài khoản`, 'info');
                logBulk(`⏱️  Delay: 1000ms giữa các tài khoản để tránh rate-limit.`, 'warn');
                logBulk('');

                let successCount = 0;
                let errorCount = 0;
                const total = validRows.length;

                for (let i = 0; i < total; i++) {
                    const row = validRows[i];
                    const pos = i + 1;

                    row._status = 'processing';
                    updatePreviewRow(row._rowIndex, 'processing', '');
                    bulkProgressLabel.textContent = `${pos}/${total}`;
                    bulkProgressText.textContent = `${pos}/${total} — Đang xử lý: ${row.email}`;

                    logBulk(`▶️  [${pos}/${total}] Đang tạo: ${row.email}`, 'info');

                    try {
                        const userCred = await firebase.auth(secondaryApp)
                            .createUserWithEmailAndPassword(row.email, row.password);

                        const newUid = userCred.user.uid;

                        try {
                            await userCred.user.updateProfile({ displayName: row.name || row.email });
                        } catch (e) {
                            console.warn('Không cập nhật được displayName:', e);
                        }

                        const dbRef = firebase.database(app).ref();
                        await dbRef.child('users/' + newUid).set({
                            email: row.email,
                            displayName: row.name || row.email,
                            role: row.role,
                            teamId: row.teamId || null,
                            createdAt: firebase.database.ServerValue.TIMESTAMP,
                            createdBy: 'bulk_import'
                        });

                        if (row.teamId) {
                            try {
                                await dbRef.child('teams/' + row.teamId + '/members/' + newUid).set(true);
                                const teamSnap = await dbRef.child('teams/' + row.teamId + '/name').once('value');
                                if (!teamSnap.exists()) {
                                    await dbRef.child('teams/' + row.teamId).update({
                                        name: row.teamId,
                                        code: row.teamId.toUpperCase(),
                                        createdAt: firebase.database.ServerValue.TIMESTAMP
                                    });
                                }
                            } catch (teamErr) {
                                console.warn('Không thêm được vào team:', teamErr);
                            }
                        }

                        try {
                            await firebase.auth(secondaryApp).signOut();
                        } catch (e) {
                            console.warn('Không signOut được secondaryApp:', e);
                        }

                        row._status = 'success';
                        row._message = 'Thành công';
                        updatePreviewRow(row._rowIndex, 'success', '');
                        successCount++;
                        logBulk(`✅ [${pos}/${total}] Thành công: ${row.name || row.email} (${row.email})`, 'success');

                    } catch (err) {
                        errorCount++;
                        row._status = 'error';

                        let msg = err.message || 'Lỗi không xác định';
                        if (err.code === 'auth/email-already-in-use') msg = 'Email đã tồn tại';
                        else if (err.code === 'auth/invalid-email') msg = 'Email không hợp lệ';
                        else if (err.code === 'auth/weak-password') msg = 'Mật khẩu quá yếu';
                        else if (err.code === 'auth/too-many-requests') msg = 'Bị chặn tạm thời (rate-limit)';
                        else if (err.code === 'PERMISSION_DENIED') msg = 'Không có quyền ghi Database';

                        row._message = msg;
                        updatePreviewRow(row._rowIndex, 'error', msg);
                        logBulk(`❌ [${pos}/${total}] Lỗi: ${msg} — ${row.email}`, 'error');

                        if (err.code === 'auth/too-many-requests') {
                            logBulk(`⏸️  Bị chặn rate-limit. Nghỉ 5 giây...`, 'warn');
                            await delay(5000);
                        }
                    }

                    const pct = Math.round((pos / total) * 100);
                    bulkProgressBar.style.width = pct + '%';

                    if (i < total - 1) {
                        await delay(1000);
                    }
                }

                bulkProgressBar.style.width = '100%';
                if (errorCount === 0) {
                    bulkProgressBar.className = 'progress-bar-inner';
                } else if (successCount === 0) {
                    bulkProgressBar.className = 'progress-bar-inner error-mode';
                } else {
                    bulkProgressBar.className = 'progress-bar-inner warning-mode';
                }
                bulkProgressLabel.textContent = 'Hoàn tất';
                bulkProgressText.textContent = `${total}/${total} — Xong`;

                logBulk('');
                logBulk('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━', 'final');
                logBulk(`🎉 HOÀN TẤT! Thành công: ${successCount}/${total} — Thất bại: ${errorCount}/${total}`, 'final');
                logBulk('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━', 'final');

                showToast(`Hoàn tất: ${successCount} thành công, ${errorCount} thất bại.`, successCount > 0 ? 'success' : 'warning', 6000);

                isBulkProcessing = false;
                btnStartBulkCreate.disabled = false;
                btnClearPreview.disabled = false;
                btnDownloadTemplate.disabled = false;
                excelDropZone.style.pointerEvents = '';
                excelDropZone.style.opacity = '';

                } catch (e) {
                    console.warn('Không refresh được danh sách user:', e);
                }
            }
